import java.security.cert.X509Certificate;

public class PivSecurityUtils {

    public static X509Certificate getClientCert() {
        VaadinServletRequest request = VaadinServletRequest.getCurrent();
        if (request != null) {
            X509Certificate[] certs = (X509Certificate[]) request.getHttpServletRequest()
                    .getAttribute("jakarta.servlet.request.X509Certificate");
            if (certs != null && certs.length > 0) {
                try {
                    certs[0].checkValidity();
                    return certs[0]; // Authenticated PIV certificate
                } catch (Exception e) {
                    // Certificate expired or invalid
                }
            }
        }
        return null;
    }
}

1. The Spring Security ConfigurationYou need to enable .x509() authentication. Since Vaadin provides its own VaadinSecurityConfigurer for view protection, you apply both.javaimport com.vaadin.flow.spring.security.VaadinSecurityConfigurer;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.security.config.annotation.web.builders.HttpSecurity;
import org.springframework.security.config.annotation.web.configuration.EnableWebSecurity;
import org.springframework.security.core.userdetails.UserDetailsService;
import org.springframework.security.web.SecurityFilterChain;

@Configuration
@EnableWebSecurity
public class SecurityConfig {

    private final X509UserDetailsService x509UserDetailsService;

    public SecurityConfig(X509UserDetailsService x509UserDetailsService) {
        this.x509UserDetailsService = x509UserDetailsService;
    }

    @Bean
    public SecurityFilterChain securityFilterChain(HttpSecurity http) throws Exception {
        // 1. Apply Vaadin default protections (CSRF, Request Cache, View Access Control)
        http.with(VaadinSecurityConfigurer.vaadin(), configurer -> {});

        // 2. Enable X.509 Pre-Authentication for PIV Cards
        http.x509(x509 -> x509
            .subjectPrincipalRegex("CN=(.*?)(?:,|$)") // Extracts the Common Name (CN) from PIV Card
            .userDetailsService(x509UserDetailsService)
        );

        return http.build();
    }
}

You must match this identifier against your database or active directory to load their roles and account status.javaimport org.springframework.security.core.authority.SimpleGrantedAuthority;

import java.util.Collections;

@Service
public class X509UserDetailsService implements UserDetailsService {

    @Override
    public UserDetails loadUserByUsername(String username) throws UsernameNotFoundException {
        // 'username' here is the CN extracted from the PIV certificate via regex
        System.out.println("Authenticating PIV Card for CN: " + username);

        // TODO: Query your database or LDAP to find the user and their roles
        if ("John Doe 1234567890".equals(username)) { 
            return new User(
                username, 
                "", // Password is unused/blank for X.509 pre-auth
                Collections.singletonList(new SimpleGrantedAuthority("ROLE_USER"))
            );
        }

        throw new UsernameNotFoundException("User not authorized with this PIV card.");
    }
}

import com.vaadin.flow.component.html.Paragraph;
import com.vaadin.flow.component.orderedlayout.VerticalLayout;
import com.vaadin.flow.router.Route;
import com.vaadin.flow.server.VaadinServletRequest;
import com.vaadin.flow.spring.security.AuthenticationContext;
import jakarta.annotation.security.PermitAll;
import java.security.cert.X509Certificate;
import java.util.Optional;

@Route("")
@PermitAll // Rejects unauthenticated PIV users automatically
public class MainView extends VerticalLayout {

    public MainView(AuthenticationContext authContext) {
        // Access standard Spring Security user details via Vaadin context
        Optional<String> username = authContext.getPrincipalName();
        
        add(new H1("Welcome to the Secure Portal"));
        username.ifPresent(name -> add(new Paragraph("Authenticated CN: " + name)));

        // Optional: Extract the raw X509 Certificate metadata if needed
        VaadinServletRequest currentRequest = VaadinServletRequest.getCurrent();
        if (currentRequest != null) {
            X509Certificate[] certs = (X509Certificate[]) currentRequest
                    .getAttribute("jakarta.servlet.request.X509Certificate");
            
            if (certs != null && certs.length > 0) {
                X509Certificate principalCert = certs[0];
                add(new Paragraph("Issuer: " + principalCert.getIssuerX500Principal().getName()));
                add(new Paragraph("Serial Number: " + principalCert.getSerialNumber()));
            }
        }
    }
}
