import org.apache.sshd.client.session.ClientSession;
import org.apache.sshd.client.channel.ChannelExec;
import java.io.ByteArrayOutputStream;
import java.io.Console;
import java.io.ByteArrayInputStream;
import java.security.KeyStore;
import java.security.Provider;
import java.security.Security;
import java.security.cert.X509Certificate;
import java.security.interfaces.RSAPrivateKey;

public class PivSshExecutor {
    public static void main(String[] args) {
        Console console = System.console();
        if (console == null) {
            System.err.println("No console available.");
            return;
        }

        // 1. Gather network & profile requirements
        String host = "192.168.1.50";
        int port = 22;
        String command = "show version"; // Example network device command

        String username = console.readLine("Enter network username: ");
        char[] pin = console.readPassword("Enter PIV Card PIN: ");

        // 2. Initialize PKCS11 Smart Card Provider
        try {
            String configString = "--name=SmartCard\nlibrary=/usr/local/lib/opensc-pkcs11.so\nslotListIndex=0";
            Provider provider = Security.getProvider("SunPKCS11");
            provider = provider.configure(configString);
            Security.addProvider(provider);

            // 3. Load Smart Card Keystore using PIN
            KeyStore keyStore = KeyStore.getInstance("PKCS11", provider);
            keyStore.load(null, pin);

            // Get the certificate alias (usually 9a or 9c for PIV cards)
            String alias = keyStore.aliases().nextElement();
            X509Certificate cert = (X509Certificate) keyStore.getCertificate(alias);
            RSAPrivateKey privateKey = (RSAPrivateKey) keyStore.getKey(alias, null);

            System.out.println("Successfully unlocked PIV Card for: " + cert.getSubjectX500Principal().getName());

            // 4. Start SSH Engine 
            try (SshClient client = SshClient.setUpDefaultClient()) {
                client.start();
                
                try (ClientSession session = client.connect(username, host, port).verify(10000).getSession()) {
                    
                    // Attach the hardware-backed asymmetric keys into the context
                    session.addPublicKeyIdentity(new java.security.KeyPair(cert.getPublicKey(), privateKey));
                    session.auth().verify(10000);
                    
                    System.out.println("SSH Authentication Successful via X.509 Identity.");

                    // 5. Run the target network command
                    try (ChannelExec channel = session.createExecChannel(command)) {
                        ByteArrayOutputStream responseStream = new ByteArrayOutputStream();
                        ByteArrayOutputStream errorStream = new ByteArrayOutputStream();
                        
                        channel.setOut(responseStream);
                        channel.setErr(errorStream);
                        channel.open().verify(5000);
                        
                        // Wait for completion
                        channel.waitFor(java.util.EnumSet.of(org.apache.sshd.client.channel.ClientChannelEvent.CLOSED), 0);
                        
                        System.out.println("\n--- COMMAND OUTPUT ---");
                        System.out.println(responseStream.toString());
                    }
                }
                client.stop();
            }

        } catch (Exception e) {
            System.err.println("Execution failed: " + e.getMessage());
            e.printStackTrace();
        }
    }
}

import org.apache.sshd.client.SshClient;
import org.apache.sshd.client.session.ClientSession;
import java.security.KeyStore;
import java.security.PrivateKey;
import java.security.cert.X509Certificate;
import java.security.KeyPair;

public class PivSshClient {
    public static void main(String[] args) {
        String host = "your.network.host";
        int port = 22;
        String username = "your_username";

        try {
            // 1. Access the Windows Certificate Store (Interactions prompt for PIV PIN here)
            KeyStore keyStore = KeyStore.getInstance("Windows-MY");
            keyStore.load(null, null); // PIN prompt triggered implicitly on cryptographic signature

            // 2. Select your PIV Card Alias (e.g., matching a thumbprint or User Principal Name)
            String alias = "your-piv-certificate-alias"; 
            X509Certificate cert = (X509Certificate) keyStore.getCertificate(alias);
            PrivateKey privateKey = (PrivateKey) keyStore.getKey(alias, null);

            if (privateKey == null || cert == null) {
                throw new IllegalStateException("PIV Key or X.509 Certificate not found in Windows-MY store.");
            }

            // Convert into a standard Java KeyPair
            KeyPair pivKeyPair = new KeyPair(cert.getPublicKey(), privateKey);

            // 3. Establish Apache MINA Client Session
            try (SshClient client = SshClient.setUpDefaultClient()) {
                client.start();
                
                try (ClientSession session = client.connect(username, host, port).verify().getSession()) {
                    // Inject the PIV X.509 public-private context
                    session.addPublicKeyIdentity(pivKeyPair);
                    
                    // Fire authentication process
                    session.auth().verify();
                    System.out.println("Successfully authenticated with PIV card via Windows-MY!");
                    
                    // Execute remote command example
                    String response = session.executeRemoteCommand("whoami");
//java -cp ".;lib/*" PivSshClient
                    System.out.println("Command Output: " + response);
                } finally {
                    client.stop();
                }
            }
        } catch (Exception e) {
            e.printStackTrace();
        }
    }
}

import javax.crypto.Cipher;
import java.util.Base64;

public static String encryptPayload(X509Certificate cert, String payload) throws Exception {
    Cipher cipher = Cipher.getInstance("RSA/ECB/PKCS1Padding");
    cipher.init(Cipher.ENCRYPT_MODE, cert.getPublicKey());
    byte[] encryptedBytes = cipher.doFinal(payload.getBytes("UTF-8"));
    return Base64.getEncoder().encodeToString(encryptedBytes);
}
